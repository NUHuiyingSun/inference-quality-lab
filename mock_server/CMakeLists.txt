cmake_minimum_required(VERSION 3.16)
project(mock_inference_server CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(mock_server src/main.cpp)
target_compile_options(mock_server PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(mock_server PRIVATE Threads::Threads)
option(ENABLE_SANITIZERS "Build with AddressSanitizer + UBSan" OFF)
if(ENABLE_SANITIZERS)
  target_compile_options(mock_server PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
  target_link_options(mock_server PRIVATE -fsanitize=address,undefined)
endif()
